"""Image -> evidence. No camera ownership, file writing, or motor commands."""
from dataclasses import dataclass  # Generates an initializer for a data container.
import cv2
import numpy as np

# Dict maps color names to lower/upper HSV tuples. OpenCV uint8 H:0..179, S/V:0..255.
# These are starting ranges to investigate, not universal color definitions.
RANGES = {"blue": ((100, 80, 50), (135, 255, 255)),
          "green": ((40, 80, 50), (85, 255, 255))}


@dataclass
class Settings:
    """Experiment parameters are chosen by us; they are not learned weights."""
    color: str = "blue"  # Annotation describes intent; value supplies a default.
    min_pixels: int = 300  # Minimum area of the largest connected colored region.
    clean: bool = False  # Optional removal of speckles using morphology.

    def __post_init__(self):
        # dataclass calls this after assigning fields. Reject invalid settings early.
        if self.color not in RANGES:
            raise ValueError("Choose blue or green")
        if type(self.min_pixels) is not int or self.min_pixels < 1:
            raise ValueError("min_pixels must be a positive integer")


def region_name(center_x, width):
    """Describe an IMAGE position, not a safe walking direction."""
    if center_x < width / 3:
        return "left"
    elif center_x < 2 * width / 3:
        return "center"
    else:
        return "right"


def analyze(bgr, settings):
    """Return (annotated_image, mask, evidence_dict) from a uint8 BGR array.

    This is a hand-written color rule, not semantic object recognition.
    """
    if bgr is None or bgr.dtype != np.uint8 or bgr.ndim != 3 or bgr.shape[2] != 3:
        raise ValueError("Expected a uint8 BGR image with three channels")
    height, width = bgr.shape[:2]  # Slice the first two dimensions; unpack the pair.
    if height == 0 or width == 0:
        raise ValueError("Image must not be empty")
    hsv = cv2.cvtColor(bgr, cv2.COLOR_BGR2HSV)  # Represent color differently.
    lower, upper = RANGES[settings.color]  # Dictionary lookup, then tuple unpacking.
    mask = cv2.inRange(hsv, np.array(lower, dtype=np.uint8),
                       np.array(upper, dtype=np.uint8))
    # Mask pixels: 255 matched all bounds; 0 did not. This is not confidence.
    if settings.clean:
        kernel = np.ones((3, 3), dtype=np.uint8)  # Neighborhood for the operation.
        mask = cv2.morphologyEx(mask, cv2.MORPH_OPEN, kernel)
        # Opening removes speckles but may also remove thin real features.
    count, labels, stats, centers = cv2.connectedComponentsWithStats(mask, connectivity=8)
    # Connected components group touching foreground pixels. Label 0 is background.
    candidates = list(range(1, count))  # [] if no foreground regions exist.
    evidence = {"color": settings.color, "found": False, "image_region": "unknown",
                "area_pixels": 0, "box": None, "center": None,
                "width": width, "height": height, "distance_cm": None}
    # None means unknown; it is not zero distance or proof that a path is clear.
    annotated = bgr.copy()  # Preserve the original pixels for comparison.
    if candidates:  # Empty lists are false; nonempty lists are true.
        # lambda defines a short ranking function; max selects the largest region.
        biggest = max(candidates, key=lambda i: stats[i, cv2.CC_STAT_AREA])
        area = int(stats[biggest, cv2.CC_STAT_AREA])  # NumPy int -> JSON-friendly int.
        evidence["area_pixels"] = area  # Keep measurement even below the threshold.
        if area >= settings.min_pixels:  # Equality is included.
            x, y, box_w, box_h, _ = [int(v) for v in stats[biggest]]
            cx, cy = [float(v) for v in centers[biggest]]  # Foreground-pixel centroid.
            evidence.update({"found": True, "image_region": region_name(cx, width),
                             "box": [x, y, x + box_w, y + box_h], "center": [cx, cy]})
            # Stored upper box coordinates are exclusive; drawing endpoints aren't.
            cv2.rectangle(annotated, (x, y), (x + box_w - 1, y + box_h - 1),
                          (0, 255, 255), 2)
            cv2.circle(annotated, (round(cx), round(cy)), 4, (0, 0, 255), -1)
    message = f"{settings.color}: {evidence['image_region']} | area={evidence['area_pixels']}"
    cv2.putText(annotated, message, (10, 25), cv2.FONT_HERSHEY_SIMPLEX,
                .6, (0, 255, 255), 2)
    return annotated, mask, evidence  # A tuple lets the caller unpack three results.


def demo_image(position="center"):
    """Known synthetic scene for a repeatable test, not a PiDog photograph."""
    frame = np.full((240, 320, 3), 190, dtype=np.uint8)  # Gray BGR canvas.
    x = {"left": 20, "center": 140, "right": 260}[position]
    frame[90:150, x:x + 30] = (255, 0, 0)  # Rows then columns; blue in BGR.
    return frame

"""Camera wrapper: owns one Pi CSI camera and returns OpenCV BGR images."""
import time  # Standard-library module for the warm-up delay.


class Camera:
    """A class is a blueprint; each instance stores its own device in self."""

    def __init__(self):
        # Delayed import lets saved-image replay work without Pi camera libraries.
        from picamera2 import Picamera2
        self.device = Picamera2()  # Create the hardware-facing library object.

    def __enter__(self):
        # Python calls this when entering `with Camera() as camera:`.
        try:
            config = self.device.create_still_configuration(
                main={"size": (640, 480)}, queue=False  # Dict containing a tuple.
            )
            self.device.configure(config)
            self.device.start()
            time.sleep(2)  # Allow exposure and white balance to begin settling.
            return self  # The variable after `as` refers to this instance.
        except BaseException:
            # __exit__ is not called if __enter__ fails; close here as well.
            self.device.close()
            raise  # Keep the original error, including an interrupted startup.

    def read(self):
        """Return uint8 pixels with shape (height, width, 3), in BGR order."""
        import cv2  # OpenCV handles image conversions.
        import numpy as np  # NumPy stores arrays of numbers.
        rgb = self.device.capture_image("main").convert("RGB")
        # Explicit Pillow RGB conversion avoids guessing camera buffer order.
        return cv2.cvtColor(np.array(rgb), cv2.COLOR_RGB2BGR)

    def __exit__(self, error_type, error_value, traceback):
        # These arguments describe an exception, if any. None does not suppress it.
        self.device.close()  # Also runs when the with-block raises an exception.

"""Connect camera -> perception -> reports. Run from the student folder with -m."""
import argparse  # Converts command-line options into Python values.
from contextlib import nullcontext  # Harmless with-block when replay needs no camera.
from dataclasses import asdict  # Settings object -> dictionary for JSON.
from datetime import datetime, timezone
import json  # Portable text format for dictionaries and lists.
from pathlib import Path  # Paths with useful methods instead of manual slashes.
import time
import cv2
from .camera import Camera  # Leading dot means another module in this package.
from .perception import Settings, analyze, demo_image


def save_image(path, pixels):
    """Finish writing before replacing a displayed image."""
    temp = path.with_name("tmp-" + path.name)
    if not cv2.imwrite(str(temp), pixels):  # OpenCV expects a filename string.
        raise OSError(f"Could not save {path}")
    temp.replace(path)  # Rename a complete file in the same directory.


def write_page(folder):
    """HTML/JavaScript display saved results; the browser does not open the camera."""
    # A triple-quoted string holds a multi-line document. No extra web library needed.
    page = '''<!doctype html><html lang="en"><meta name="viewport" content="width=device-width,initial-scale=1">
<title>PiDog observations</title><style>body{font:18px system-ui;margin:24px;background:#eef5f2;color:#143f43}img{width:min(100%,640px);display:block;margin-bottom:20px}pre{white-space:pre-wrap}</style>
<h1>PiDog observations</h1><p>Latest saved observation, not live video. Distance is unknown. No motor commands. Check the timestamp: refresh continues after capture ends. Images and JSON can briefly come from adjacent frames while files update.</p>
<h2>Original</h2><img id="original" alt="Original"><h2>Color mask</h2><img id="mask" alt="Mask"><h2>Annotated</h2><img id="annotated" alt="Annotated"><pre id="record"></pre>
<script>async function update(){const q=Date.now();for(const n of ['original','mask','annotated'])document.getElementById(n).src=n+'.png?t='+q;
try{const r=await fetch('latest.json?t='+q);document.getElementById('record').textContent=JSON.stringify(await r.json(),null,2)}catch{document.getElementById('record').textContent='Open latest.json separately if viewing from a file.'}}update();setInterval(update,3000);</script></html>'''
    (folder / "index.html").write_text(page, encoding="utf-8")


def main():
    """Read settings, own the camera once, process a bounded sequence, then close."""
    parser = argparse.ArgumentParser(description=__doc__)
    source = parser.add_mutually_exclusive_group(required=True)  # Choose exactly one.
    source.add_argument("--camera", action="store_true")  # A Boolean flag.
    source.add_argument("--image", type=Path)  # Repeatable saved-image experiment.
    source.add_argument("--demo", action="store_true")  # No camera needed.
    parser.add_argument("--out", type=Path, required=True)
    parser.add_argument("--color", choices=["blue", "green"], default="blue")
    parser.add_argument("--min-pixels", type=int, default=300)
    parser.add_argument("--clean", action="store_true")
    parser.add_argument("--frames", type=int, default=1)
    parser.add_argument("--interval", type=float, default=1.0)
    args = parser.parse_args()  # --help is provided automatically.
    if not 1 <= args.frames <= 600 or not 0 <= args.interval <= 10:
        parser.error("Use 1..600 frames and 0..10 seconds between observations")
    if args.image and args.frames != 1:
        parser.error("Saved-image replay uses --frames 1")
    if args.min_pixels < 1:
        parser.error("min-pixels must be positive")
    settings = Settings(args.color, args.min_pixels, args.clean)
    args.out.mkdir(parents=True, exist_ok=False)  # Never erase an earlier trial.
    owner = Camera() if args.camera else nullcontext(None)  # Conditional expression.
    with owner as camera:
        for number in range(args.frames):  # Finite loop: 0 through frames-1.
            if args.camera:
                frame = camera.read()
            elif args.image:
                frame = cv2.imread(str(args.image))  # imread already returns BGR.
                if frame is None:
                    raise ValueError(f"Could not read image: {args.image}")
            else:
                frame = demo_image(["left", "center", "right"][number % 3])
                # Modulo cycles through three positions in synthetic mode.
            start = time.perf_counter()  # Monotonic clock for measuring duration.
            annotated, mask, record = analyze(frame, settings)
            record["processing_ms"] = round(1000 * (time.perf_counter() - start), 2)
            record["frame"] = number
            record["timestamp_utc"] = datetime.now(timezone.utc).isoformat()
            record["source"] = "camera" if args.camera else ("image" if args.image else "synthetic")
            record["settings"] = asdict(settings)
            save_image(args.out / "original.png", frame)
            save_image(args.out / "mask.png", mask)
            save_image(args.out / "annotated.png", annotated)
            # Keep latest images only to limit SD-card writes; retain all text records.
            temporary = args.out / "tmp-latest.json"
            temporary.write_text(json.dumps(record, indent=2), encoding="utf-8")
            temporary.replace(args.out / "latest.json")
            with (args.out / "history.jsonl").open("a", encoding="utf-8") as log:
                log.write(json.dumps(record) + "\n")  # One JSON dictionary per line.
            if number == 0:
                write_page(args.out)  # Viewer appears after its first images exist.
            print(f"{number}: {record['image_region']} | area {record['area_pixels']}")
            if number + 1 < args.frames:
                time.sleep(args.interval)  # Delay PLUS work; not a guaranteed frame rate.


if __name__ == "__main__":
    # Importing this module defines functions. Running it calls main exactly once.
    try:
        main()
    except KeyboardInterrupt:
        print("Stopped; camera context closed. Completed observations are saved.")
